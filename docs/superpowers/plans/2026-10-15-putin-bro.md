# Broforce: Putin, the second bro — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A second playable parody bro, Vladimir Putin: BroLee's sheet repainted, a judo throw, an exploding
matryoshka, a menu choice of bro, and a swap to the other bro on every rescue.

**Architecture:** The extractor gains BroLee's prefab values and a second repaint recipe with a hair mask. In the
engine, a bro's unit values come from its `base` prefab. `World` holds a roster and swaps bro in place on a rescue.
Weapons and specials become kit handlers (`engine/kits.ts`): the judo throw is a melee state ported from BroLee,
and the grabbed mook flies off as a `body` projectile. The matryoshka is a `doll` projectile that opens into three
`dollSmall` ones. The renderer reads the player's sheet from the world every frame. The menu picks the starting bro.

**Tech Stack:** Python 3.12, UnityPy 1.25.4, Pillow, pytest · Node 22, React 18.3.1, Vite 6.0.7, TypeScript 5.7.2
(`strict`, `noUnusedLocals`, `noUnusedParameters`), Vitest 2.1.8.

```python
"""Bro repaints: deterministic colour remaps of an original bro's body sheet.

Brolusconi repaints Double Bro Seven (palette measured from 00Bro7_anim, 29 colours). Putin repaints BroLee
(13 colours), whose hair and trousers share one black, so a hair mask tells them apart first. Colours not
listed are kept.
"""
from dataclasses import dataclass

from PIL import Image

RGB = tuple[int, int, int]

PALETTE_MAP: dict[RGB, RGB] = {
    # tuxedo -> navy suit
    (0x20, 0x20, 0x20): (0x1F, 0x2A, 0x4A),
    (0x2B, 0x2D, 0x2F): (0x2A, 0x37, 0x5E),
    (0x1F, 0x1D, 0x1D): (0x18, 0x21, 0x3B),
    # brown hair -> slicked near-black with a shine
    (0x20, 0x13, 0x06): (0x14, 0x0E, 0x0A),
    (0x3E, 0x2B, 0x1A): (0x3A, 0x30, 0x2A),
    (0x2C, 0x1A, 0x09): (0x1E, 0x16, 0x10),
    # skin -> tan
    (0xCF, 0x8C, 0x71): (0xD8, 0x8A, 0x52),
    (0xE9, 0xA6, 0x8C): (0xF0, 0xA8, 0x6C),
    (0xAA, 0x73, 0x5D): (0xAE, 0x68, 0x3A),
    (0xA8, 0x86, 0x78): (0xB0, 0x80, 0x5C),
    # red flower -> blue tie
    (0x7F, 0x11, 0x00): (0x1E, 0x3A, 0x8A),
    (0x6B, 0x17, 0x0B): (0x17, 0x2E, 0x6E),
    (0xA1, 0x04, 0x03): (0x2D, 0x4F, 0xB0),
    (0x95, 0x26, 0x1A): (0x26, 0x45, 0x9C),
    (0xA4, 0x0E, 0x0C): (0x2D, 0x4F, 0xB0),
}

# BroLee's tan skin -> pale pink
PUTIN_MAP: dict[RGB, RGB] = {
    (0xEE, 0xB8, 0x91): (0xF4, 0xC9, 0xB4),
    (0xD7, 0xA1, 0x7F): (0xE0, 0xAD, 0x98),
    (0xC3, 0x86, 0x6D): (0xC8, 0x93, 0x82),
    (0xAA, 0x73, 0x5D): (0xA9, 0x77, 0x69),
    (0xAA, 0x7C, 0x5D): (0xAD, 0x80, 0x6E),
}


@dataclass(frozen=True)
class HairMask:
    """Splits a colour shared by hair and trousers. In each frame, a region of `base` + `shade` pixels
    (8-connected, grown from a `base` pixel) that touches a `feet` pixel is trousers; any other is hair."""
    base: RGB
    shade: RGB
    feet: frozenset
    hair: RGB
    hair_shade: RGB
    trousers: RGB
    frame: int = 32


# BroLee: hair top and trousers #161616, black sides/back of the head, belt and trouser shading, shoes #0E0E0E,
# socks #F1F2F6 -> ash-brown hair with darker sides, olive trousers (black shading, shoes and socks kept)
PUTIN_HAIR = HairMask(
    base=(0x16, 0x16, 0x16), shade=(0x00, 0x00, 0x00), feet=frozenset({(0x0E, 0x0E, 0x0E), (0xF1, 0xF2, 0xF6)}),
    hair=(0xA8, 0x8B, 0x6A), hair_shade=(0x6B, 0x54, 0x40), trousers=(0x4B, 0x53, 0x20),
)


@dataclass(frozen=True)
class Recipe:
    key: str                  # output sheet: sprites/<key>.png
    source: str               # catalog SpriteRef key of the sheet it repaints
    palette: dict
    hair: HairMask | None = None


RECIPES = (
    Recipe("brolusconi", "bro_base", PALETTE_MAP),
    Recipe("putin", "bro_lee", PUTIN_MAP, PUTIN_HAIR),
)


def _split_hair(px, width: int, height: int, mask: HairMask) -> tuple[set, set]:
    """(hair pixels, trousers pixels) of every base/shade region; a region never crosses a frame edge."""
    hair: set = set()
    trousers: set = set()
    seen: set = set()

    def colour(x, y):
        r, g, b, a = px[x, y]
        return (r, g, b) if a else None

    for fy in range(0, height, mask.frame):
        for fx in range(0, width, mask.frame):
            x1, y1 = min(fx + mask.frame, width), min(fy + mask.frame, height)
            for y in range(fy, y1):
                for x in range(fx, x1):
                    if (x, y) in seen or colour(x, y) != mask.base:
                        continue
                    region, stack, feet = [], [(x, y)], False
                    seen.add((x, y))
                    while stack:
                        cx, cy = stack.pop()
                        region.append((cx, cy))
                        for nx in (cx - 1, cx, cx + 1):
                            for ny in (cy - 1, cy, cy + 1):
                                if not (fx <= nx < x1 and fy <= ny < y1):
                                    continue
                                c = colour(nx, ny)
                                if c in mask.feet:
                                    feet = True
                                elif (nx, ny) not in seen and c in (mask.base, mask.shade):
                                    seen.add((nx, ny))
                                    stack.append((nx, ny))
                    (trousers if feet else hair).update(region)
    return hair, trousers


def repaint(img: Image.Image, mapping: dict = PALETTE_MAP, hair: HairMask | None = None) -> Image.Image:
    out = img.convert("RGBA").copy()
    px = out.load()
    width, height = out.size
    hair_px, trousers_px = _split_hair(px, width, height, hair) if hair else (set(), set())
    for y in range(height):
        for x in range(width):
            r, g, b, a = px[x, y]
            if a == 0:
                continue
            new = mapping.get((r, g, b))
            if hair and (x, y) in hair_px:
                new = hair.hair if (r, g, b) == hair.base else hair.hair_shade
            elif hair and (x, y) in trousers_px and (r, g, b) == hair.base:
                new = hair.trousers
            if new is not None:
                px[x, y] = (*new, a)
    return out
```

```ts
import {
  JUDO_FRAME_RATE, JUDO_FRAMES, JUDO_GROUND, JUDO_HIT, JUDO_STRIKE_FRAME, JUDO_STRIKE_LAST, JUDO_THROW, TILE,
  UNIT_BOX_HEIGHT, UNIT_WIDTH,
} from './constants';
import { barrelBlast, type Blast } from './explosions';
import { getColumn, getRow } from './map';
import { throwBody } from './projectiles';
import type { EnemyKind, Unit } from './types';
import type { World } from './world';

/** BroLee.StartFiring, forward: a new strike, alternating between the two strike animations. */
export function startJudo(p: Unit): void {
  p.attackRow = p.attackRow === 0 ? 1 : 0;
  p.attack = { frame: 0, counter: 0, row: p.attackRow, hit: false, struck: false };
  p.flex = null;
}

/** Map.HitUnits' box test (the one Map.AlertNearbyMooks uses too). */
function inHitBox(u: Unit, x: number, y: number, range: number): boolean {
  return Math.abs(u.x - x) - range < UNIT_WIDTH && Math.abs(u.y + UNIT_BOX_HEIGHT / 2 + 3 - y) - range < UNIT_BOX_HEIGHT;
}

/** The first live enemy in BroLee's strike box. */
export function judoTarget(enemies: readonly Unit[], p: Unit): Unit | null {
  const x = p.x + p.facing * JUDO_HIT.x;
  const y = p.y + JUDO_HIT.y;
  return enemies.find((m) => !m.dead && m.invulnerable <= 0 && inHitBox(m, x, y, JUDO_HIT.range)) ?? null;
}

/**
 * The grab (ours, in place of BroLee's BackSomersault): the mook leaves the fight as a kill and flies off as a
 * body. It's marked gibbed so it vanishes from the map, and exploded so a suicide mook doesn't go off in the
 * bro's hands (MookSuicide.Gib): its body carries the blast.
 */
function grab(w: World, p: Unit, m: Unit): void {
  const kind = m.kind as EnemyKind;
  w.events.push({ type: 'kill', x: m.x, y: m.y, kind, xI: p.facing * JUDO_THROW.x, yI: JUDO_THROW.y });
  if (m.originDoor >= 0) w.doors[m.originDoor].mookCount--; // Mook.Death: MookDoor.RemoveMook
  m.originDoor = -1;
  m.dead = true;
  m.deadTime = 0;
  m.gibbed = true;
  m.reported = { death: true, gib: true };
  if (m.ai) m.ai.exploded = true;
  w.projectiles.push(throwBody(w.nextId++, p, kind));
  w.events.push({ type: 'shot', weapon: 'body', x: p.x, y: p.y, facing: p.facing });
}

/** FireWeaponGround: the first block within reach ahead takes groundSwordDamage. True when one was struck. */
function strikeGround(w: World, p: Unit, blasts: Blast[]): boolean {
  const y = p.y + JUDO_GROUND.y;
  for (let d = 0; d <= JUDO_GROUND.reach; d++) {
    const x = p.x + p.facing * (JUDO_GROUND.x + d);
    if (!w.map.blockAt(x, y)) continue;
    const c = getColumn(x);
    const r = getRow(y);
    const kind = w.map.damage(c, r, JUDO_GROUND.damage);
    if (kind) w.events.push({ type: 'block', x: c * TILE, y: r * TILE, kind, damage: JUDO_GROUND.damage });
    if (kind === 'barrel') blasts.push(barrelBlast(c * TILE, r * TILE, w.stats.barrelRange));
    return true;
  }
  return false;
}

/** BroLee.RunFiring + AnimateAttackForwards for one fixed step. */
export function stepJudo(w: World, p: Unit, t: number, blasts: Blast[]): void {
  const a = p.attack;
  if (!a) return;
  a.counter += t;
  while (a.counter >= JUDO_FRAME_RATE) {
    a.counter -= JUDO_FRAME_RATE;
    a.frame++;
  }
  if (a.frame >= JUDO_FRAMES) {
    p.attack = null;
    return;
  }
  if (a.hit || a.frame < JUDO_STRIKE_FRAME) return;
  if (a.frame <= JUDO_STRIKE_LAST) {
    const m = judoTarget(w.enemies, p);
    if (m) {
      grab(w, p, m);
      a.hit = true;
      return;
    }
  }
  if (!a.struck) {
    a.struck = true;
    a.hit = strikeGround(w, p, blasts);
  }
}
```

```ts
/** The matryoshka's two half shells flying apart when it opens: our own art, like the football. */
export interface Shell {
  x: number;
  y: number;
  xI: number;
  yI: number;
  age: number;
  side: -1 | 1; // which half
}

export const SHELL_LIFE = 0.6;
const SHELL_GRAVITY = 600;

export function openShells(x: number, y: number): Shell[] {
  return ([-1, 1] as const).map((side) => ({ x, y, xI: side * 70, yI: 120, age: 0, side }));
}

export function stepShells(shells: Shell[], dt: number): Shell[] {
  for (const s of shells) {
    s.age += dt;
    s.yI -= SHELL_GRAVITY * dt;
    s.x += s.xI * dt;
    s.y += s.yI * dt;
  }
  return shells.filter((s) => s.age < SHELL_LIFE);
}
```

```ts
/** The installed voice for a language: the exact tag, else any voice of the language (it-CH -> it-IT). */
export function pickVoice<V extends { lang: string }>(voices: readonly V[], lang: string): V | undefined {
  const prefix = lang.split('-')[0].toLowerCase();
  return voices.find((v) => v.lang === lang) ?? voices.find((v) => v.lang.toLowerCase().startsWith(prefix));
}

/** Says the line with the browser's generic voice for the bro's language (no recording of the real person). */
export function speak(text: string, lang: string, enabled: boolean): void {
  if (!enabled || typeof window === 'undefined' || !('speechSynthesis' in window)) return;
  const utterance = new SpeechSynthesisUtterance(text);
  utterance.lang = lang;
  utterance.rate = 1.05;
  utterance.pitch = 0.9;
  const voice = pickVoice(window.speechSynthesis.getVoices(), lang);
  if (voice) utterance.voice = voice;
  window.speechSynthesis.cancel();
  window.speechSynthesis.speak(utterance);
}
```
